# Widdlemon SMP — host settings the pack can't manage (set these in the host panel)
# Host: Pterodactyl, 16 GB container, 4 vCPU (AMD Ryzen 9 9950X), Java 21.

# ---------------------------------------------------------------------------------------------
# 1) Startup command (Pterodactyl "Startup" tab). 12 GB heap leaves ~4 GB for the JVM's
#    off-heap memory (156 mods' classes, JIT code, threads, network buffers) — using the full
#    16 GB risks the container being OOM-killed with no crash report. Aikar's G1 flags (>12 GB).
# ---------------------------------------------------------------------------------------------
java -Xms12G -Xmx12G -XX:+UseContainerSupport -XX:+UseG1GC -XX:+ParallelRefProcEnabled -XX:MaxGCPauseMillis=200 -XX:+UnlockExperimentalVMOptions -XX:+DisableExplicitGC -XX:+AlwaysPreTouch -XX:G1NewSizePercent=40 -XX:G1MaxNewSizePercent=50 -XX:G1HeapRegionSize=16M -XX:G1ReservePercent=15 -XX:G1HeapWastePercent=5 -XX:G1MixedGCCountTarget=4 -XX:InitiatingHeapOccupancyPercent=20 -XX:G1MixedGCLiveThresholdPercent=90 -XX:G1RSetUpdatingPauseTimePercent=5 -XX:SurvivorRatio=32 -XX:+PerfDisableSharedMem -XX:MaxTenuringThreshold=1 -Dusing.aikars.flags=https://mcflags.emc.gs -Daikars.new.flags=true -jar fabric-server-launch.jar nogui

# OUR HOST: the startup command can't be edited, only "additional flags". Paste this into that field.
# Java uses the LAST -Xmx it sees: check the command echoed at the top of the console on boot —
# -Xmx12G must appear AFTER the panel's -Xmx16000M. If it appears before, lower the server's memory
# setting in the panel (or ask the host to) to ~12 GB instead.
-Xms12G -Xmx12G -XX:+UseG1GC -XX:+ParallelRefProcEnabled -XX:MaxGCPauseMillis=200 -XX:+UnlockExperimentalVMOptions -XX:+DisableExplicitGC -XX:+AlwaysPreTouch -XX:G1NewSizePercent=40 -XX:G1MaxNewSizePercent=50 -XX:G1HeapRegionSize=16M -XX:G1ReservePercent=15 -XX:G1HeapWastePercent=5 -XX:G1MixedGCCountTarget=4 -XX:InitiatingHeapOccupancyPercent=20 -XX:G1MixedGCLiveThresholdPercent=90 -XX:G1RSetUpdatingPauseTimePercent=5 -XX:SurvivorRatio=32 -XX:+PerfDisableSharedMem -XX:MaxTenuringThreshold=1 -Dusing.aikars.flags=https://mcflags.emc.gs -Daikars.new.flags=true
